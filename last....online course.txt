      return `

    /* Search / Filter / Sort */
    const searchInput = document.getElementById('search');
    const filterType = document.getElementById('filterType');
    const sortBy = document.getElementById('sortBy');
    const resetFilters = document.getElementById('resetFilters');

    function applyFilters(){
      const q = searchInput.value.toLowerCase().trim();
      const type = filterType.value;
      const sort = sortBy.value;

      function filterItems(list){
        return list.filter(i=>{
          const matchText = i.title.toLowerCase().includes(q) ||
                            i.desc.toLowerCase().includes(q) ||
                            i.features.some(f=> f.toLowerCase().includes(q));
          const matchType = (type === 'all') ? true : i.type === type;
          return matchText && matchType;
        }).sort((a,b)=>{
          return (sort === 'az') ? a.title.localeCompare(b.title) : b.title.localeCompare(a.title);
        });
      }

      const filteredCourses = filterItems([...COURSES]);
      const filteredWorkshops = filterItems([...WORKSHOPS]);

      coursesGrid.innerHTML = '';
      workshopsGrid.innerHTML = '';
      filteredCourses.forEach(c=> coursesGrid.appendChild(createCard(c)));
      filteredWorkshops.forEach(w=> workshopsGrid.appendChild(createCard(w)));
    }

    /* IndexedDB backend (local) */
    const DB_NAME = 'portalDB';
    const STORE = 'registrations';
    let db;

    function openDB(){
      return new Promise((resolve, reject)=>{
        const req = indexedDB.open(DB_NAME, 1);
        req.onupgradeneeded = (e)=>{
          const db = e.target.result;
          if(!db.objectStoreNames.contains(STORE)){
            db.createObjectStore(STORE, { keyPath: 'id', autoIncrement: true });
          }
        };
        req.onsuccess = ()=> resolve(req.result);
        req.onerror = ()=> reject(req.error);
      });
    }

    async function saveRegistration(reg){
      if(!db) db = await openDB();
      return new Promise((resolve, reject)=>{
        const tx = db.transaction(STORE, 'readwrite');
        tx.objectStore(STORE).add(reg);
        tx.oncomplete = ()=> resolve();
        tx.onerror = ()=> reject(tx.error);
      });
    }

    async function getAllRegistrations(){
      if(!db) db = await openDB();
      return new Promise((resolve, reject)=>{
        const tx = db.transaction(STORE, 'readonly');
        const req = tx.objectStore(STORE).getAll();
        req.onsuccess = ()=> resolve(req.result || []);
        req.onerror = ()=> reject(req.error);
      });
    }

    async function clearRegistrations(){
      if(!db) db = await openDB();
      return new Promise((resolve, reject)=>{
        const tx = db.transaction(STORE, 'readwrite');
        const req = tx.objectStore(STORE).clear();
        req.onsuccess = ()=> resolve();
        req.onerror = ()=> reject(req.error);
      });
    }

    /* Optional API sync (/api/register) */
    async function trySyncToAPI(payload){
      try{
        const res = await fetch('/api/register', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        return res.ok;
      }catch(err){
        return false;
      }
    }

    function isValidEmail(email){
      return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    }

      const name = document.getElementById('name').value.trim();

      closeModal();
    });

    function toCSV(rows){
      const headers = ['name','email','phone','program','type','notes','time'];
      const lines = [headers.join(',')];
      rows.forEach(r=>{
        const line = headers.map(h=> `"${String(r[h] ?? '').replace(/"/g,'""')}"`).join(',');
        lines.push(line);
      });
      return lines.join('\n');
    }

    function escapeHtml(str){
      return String(str).replace(/[&<>"']/g, s => ({
        '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
      })[s]);
    }

    /* Glitter */
    const canvas = document.getElementById('glitter');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function resizeCanvas(){
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      particles = createParticles(150);
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    function createParticles(n){
      const colors = [ '#ffffff', '#ffeaa7', '#a29bfe', '#81ecec', '#55efc4', '#fab1a0', '#ff9ff3' ];
      const arr = [];
      for(let i=0;i<n;i++){
        arr.push({
          x: Math.random()*canvas.width,
          y: Math.random()*canvas.height,
          r: Math.random()*1.8+0.6,
          dx: (Math.random()-0.5)*0.6,
          dy: (Math.random()-0.5)*0.6,
          alpha: Math.random()*0.8+0.2,
          color: colors[Math.floor(Math.random()*colors.length)]
        });
      }
      return arr;
    }
    function drawParticles(){
      ctx.clearRect(0,0,canvas.width,canvas.height);
      particles.forEach(p=>{
        p.x += p.dx; p.y += p.dy;
        if(p.x<0 || p.x>canvas.width) p.dx *= -1;
        if(p.y<0 || p.y>canvas.height) p.dy *= -1;
        ctx.globalAlpha = p.alpha;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI*2);
        ctx.fillStyle = p.color;
        ctx.fill();
      });
      requestAnimationFrame(drawParticles);
    }
    drawParticles();

      overlay.style.position = 'fixed';
      overlay.style.inset = '0';
      overlay.style.pointerEvents = 'none';
      overlay.style.zIndex = '9999';
      document.body.appendChild(overlay);

      for(let i=0;i<count;i++){
        const conf = document.createElement('span');
        const size = Math.random()*8+4;
        conf.style.position = 'absolute';
        conf.style.left = Math.random()*100 + '%';
        conf.style.top = '-10px';
        conf.style.width = size + 'px';
        conf.style.height = size + 'px';
        conf.style.borderRadius = '2px';
        conf.style.background = ['#ff6b6b','#48dbfb','#5f27cd','#10ac84','#feca57','#ff9ff3'][i%6];
        conf.style.transform = `rotate(${Math.random()*360}deg)`;
        conf.style.opacity = '0.9';
        conf.style.filter = 'saturate(140%)';
        overlay.appendChild(conf);

        const duration = Math.random()*1500 + 1800;
        const translateX = (Math.random()-0.5)*120;
        conf.animate([
          { transform: conf.style.transform, top: '-10px' },
          { transform: `translate(${translateX}px, ${window.innerHeight * 0.9}px) rotate(${Math.random()*360}deg)`, top: '100%' }
        ], { duration, easing: 'cubic-bezier(.2,.8,.2,1)', fill: 'forwards' });
      }

      setTimeout(()=> document.body.removeChild(overlay), 2300);
    }
